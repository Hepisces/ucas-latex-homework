\question
\begin{problem}
用数学归纳法证明 $\sum_{i=1}^{n}i=n(n+1)/2$。
\end{problem}
\solution
% induction 是归纳证明环境，三个命令分别标出基础、假设和归纳步骤。
\begin{induction}
\basecase 当 $n=1$ 时，两边均为 $1$。
\indhyp 假设 $n=k$ 时，$\sum_{i=1}^{k}i=k(k+1)/2$。
\indstep 当 $n=k+1$ 时，
\[
\sum_{i=1}^{k+1}i=\frac{k(k+1)}2+(k+1)=\frac{(k+1)(k+2)}2.
\]
\end{induction}
